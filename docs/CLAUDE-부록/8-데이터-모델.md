# CLAUDE.md 부록 — 8. 데이터 모델

CLAUDE.md 에서 2026-10-07 에 옮긴 본문이다. 절 번호와 절 머리는 CLAUDE.md 와 같고,
옮길 때 본문은 한 글자도 바꾸지 않았다. 이 문서가 그 장의 정본이다.

       loginId?, passwordHash?,        // **아이디는 이름의 로마자다** — 전화번호를
                                       // 아이디로 쓰지 않는다(화면·기록·DB 에 번호가
                                       // 남고, 번호가 바뀌면 아이디까지 바꿔야 한다).
                                       // phoneNumber 는 연락처로 남는다.
                                       // 겹치면 안 되는 것은 loginId 라 유니크가
                                       // 이쪽이다 — ix_users_login_id (빈 값끼리는
                                       // 겹침이 아니다).
                                       // passwordHash 한 줄에 **셈(n·r·p)이 함께**
                                       // 있고 읽을 때 그 셈으로 센다 (domain/login)
       mustChangePassword,             // 관리자가 발급한 첫 비밀번호인가. 참이면
                                       // **인증 의존 한 곳**에서 막고 바꾸는 화면으로
                                       // 보낸다 — 화면마다 걸지 않는다 (4-12)
       failedCount, lockedUntil?,      // 10번 틀리면 5분. **DB 에 둔다** — 메모리면
                                       // 서버를 다시 켜는 것으로 풀린다
       firstSeenAt? }                  // 이 계정으로 처음 화면을 연 때 (4-16).
                                       // 계정을 만든 때와 다르다 — 총무팀이 먼저
                                       // 만들어 두고 비밀번호는 나중에 전한다.
                                       // **사이드바 배지가 이 값을 경계로 센다** —
                                       // 들어오기 전에 쌓인 것은 세지 않는다.
                                       // 비어 있으면 아직 한 번도 안 들어온 것이고,
                                       // 처음 들어오는 그 요청이 찍는다

TaskLibrary {
  id, title, kind,                    // 'main' | 'sub' | 'schedule'
  parentLibraryId?,
  defaultDepartmentId, defaultDWeek,
  datesChangedAt?,                    // 상대 위치를 마지막으로 고친 때 (6-4).

                                      // 고칠 때는 안 찍는다
  relatedLibraryIds[],                // 관련업무 — 방향 없음. 양쪽에 서로 적는다
  prerequisiteLibraryIds[],           // 선행업무 — 방향 있음. 가진 쪽에만 적는다.
                                      // 후속("나를 기다리는 업무")은 저장하지 않고 조회할 때 계산한다
  relatedDepartmentIds[],             // 관련팀 (고스트 표시 대상)
  alwaysRequired: boolean,            // 총무팀이 손으로 지정한 '매 회차 반드시'
  rules?,                             // 업무 규칙 — 회차를 넘어가는 진행 방식 (4-9)
  origin,                             // 'history' | 'claude_suggestion'
  suggestionRationale?,               // 제안 근거 (origin이 제안일 때 필수)
  createdAt, archivedAt?
}

TaskRun {
  id, libraryId, retreatId,
  included: boolean,                  // false여도 삭제하지 않고 기록
  runNo,                              // 회차 안에서 한 번 매기고 안 바뀌는 번호 (4-14)
                                      // — 회의에서 「12번」 으로 부른다. 새 run 은 max+1,
                                      // 새 회차는 1부터. 빠진 번호는 메우지 않는다
  departmentId, assigneeId?,
  dWeek, startDate, endDate,
  status,                             // '대기'|'진행중'|'완료'  (models.RUN_STATUSES)
                                      // '지연' 은 저장하지 않는다 — 계산값이다 (4-3)
  startedAt?,                         // 처음 '대기'를 벗어난 날. 되돌려도 지우지 않는다 (4-10)
  completedAt?,                       // 실제로 끝난 날. '완료'를 벗어나면 지운다
                                      // (착수는 사실이지만 완료는 취소된다)
  blockedByRunIds[],
  movedOffsetDays?,                   // 사람이 보드에서 옮긴 자리 — **날짜가 아니라
                                      // 「그때 셈한 자리에서 며칠」** 이다 (6-4).
                                      // NULL 이면 안 옮긴 것. 0 도 값이다
  movedAt?,                           // 그 자리를 정한 때. 라이브러리의
                                      // datesChangedAt 과 견주어 **나중 것이 이긴다**.
                                      // 비면 지금은 안 따르는 것이고 위 값은 남는다 —
                                      // 물러선 쪽을 지우지 않는 자리다 (0장)
  sourceMeetingId?                    // 출처 — 회의 항목에서 전환됐거나 제안을
                                      // 반영해 만든 업무면 그 회의록 (12장).
                                      // 논의의 sourceMeetingId(4-9)와 같은
                                      // 자리 — 논의는 논의 쪽에, 업무는 업무
                                      // 쪽에 단다. 비면 사람이 보드·마법사에서
                                      // 만든 것.
                                      // **「이미 만든 제안인가」 를 이것으로
                                      // 센다** — (출처, 제목)이 그 답이고
                                      // 따로 저장하지 않는다 (12장 · 14장의
                                      // made_from_meeting)
}
